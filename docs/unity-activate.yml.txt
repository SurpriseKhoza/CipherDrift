name: Unity license activation (Personal)

# Run manually from the Actions tab. Saves the result as the UNITY_LICENSE secret.
# Needs repo secrets: UNITY_EMAIL, UNITY_PASSWORD, REPO_ACCESS_TOKEN
# REPO_ACCESS_TOKEN = a GitHub personal access token that can write this repo's secrets.

on:
  workflow_dispatch:

jobs:
  activate:
    runs-on: ubuntu-latest
    steps:
      - name: Request activation file (.alf)
        id: alf
        uses: game-ci/unity-request-activation-file@v2
        with:
          unityVersion: 6000.3.0f1

      - uses: actions/setup-node@v4
        with:
          node-version: '20'

      - name: Install activation tool
        run: npm install --global unity-license-activate

      - name: Activate and save UNITY_LICENSE secret
        env:
          UNITY_EMAIL: ${{ secrets.UNITY_EMAIL }}
          UNITY_PASSWORD: ${{ secrets.UNITY_PASSWORD }}
          ALF: ${{ steps.alf.outputs.filePath }}
          GH_TOKEN: ${{ secrets.REPO_ACCESS_TOKEN }}
        run: |
          set -euo pipefail
          unity-license-activate "$UNITY_EMAIL" "$UNITY_PASSWORD" "$ALF"
          ulf="$(ls -1 *.ulf | head -n 1)"
          test -n "$ulf"
          # Never print the license contents: logs can be public.
          gh secret set UNITY_LICENSE --repo "$GITHUB_REPOSITORY" < "$ulf"
          echo "UNITY_LICENSE secret updated."
